package org.javier.tarea2;
import java.io.FileOutputStream;
import java.io.IOException;
import java.io.RandomAccessFile;

public class Ejercicio_1  {
    static void main() {
        //PARTE 1. ESCRIBIR FICHERO
        String[] apellido = {"Roman", "Frances", "Garcia", "Robin"};
        int[] departamento = {23, 45, 36, 15};
        double[] salario = {1523.23, 1235.56, 1856.89, 2450.23};
        try(RandomAccessFile raf = new RandomAccessFile("C:\\Users\\Javier\\Desktop\\AccesoDatosEjercicios\\Ejercicio 2\\empleados.txt", "rw")){
            //VACIAMOS EL FICHERO -> raf.setLength(0)
            raf.setLength(0);
            for(int i = 0; i < apellido.length; i ++){
                raf.writeInt(i+1);         //Tamaño 4Bytes
                StringBuffer sb = new StringBuffer(apellido[i]);
                //Creamos un objeto para poder modificar posteriormente
                //el tamaño del String
                sb.setLength(10);
                raf.writeChars(sb.toString());//Tamaño 10. 10x2 = 20 Bytes
                raf.writeInt(departamento[i]);//Tamaño 4 Bytes
                raf.writeDouble(salario[i]);  //Tamaño 8 Bytes
            }                                 //Total  36 Bytes
            System.out.println("Fichero binario creado con éxito.");
            raf.seek(0);

            //PARTE 2. LEER FICHERO
            char[] apellidoLeer = new char[10];
            String apellidoStr;
            int departamentoLeer;
            double salarioLeer;
            int id;

            while(raf.getFilePointer() < raf.length()){
            id = raf.readInt();
            for(int i = 0; i < 10; i++){
                apellidoLeer[i] = raf.readChar();
            }
            apellidoStr = new String(apellidoLeer).trim();
            departamentoLeer = raf.readInt();
            salarioLeer = raf.readDouble();
            System.out.println(id + "-" +  apellidoStr + "-" + departamentoLeer + "-" + salarioLeer);
        }
        }catch(IOException e){
            e.printStackTrace();
        }
    }
}
